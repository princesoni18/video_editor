# ─────────────────────────────────────────────────────────────────────────────
#  HELPERS
# ─────────────────────────────────────────────────────────────────────────────

def _run(cmd: list, label: str, allow_fail: bool = False):
    log.info(f"[ffmpeg:{label}] " + " ".join(str(c) for c in cmd[:7]) + " ...")
    r = subprocess.run([str(c) for c in cmd], capture_output=True, text=True)
    if r.returncode != 0:
        log.error(f"[ffmpeg:{label}] STDERR:\n{r.stderr[-2000:]}")
        if not allow_fail:
            raise RuntimeError(f"FFmpeg step '{label}' failed")
    return r


def _probe(path: str) -> tuple[int, int, float]:
    """Returns (width, height, duration_seconds)"""
    cmd = ["ffprobe", "-v", "quiet", "-print_format", "json",
           "-show_streams", "-show_format", path]
    data = json.loads(subprocess.check_output(cmd))
    w, h, dur = 1080, 1920, 0.0
    for s in data.get("streams", []):
        if s.get("codec_type") == "video":
            w   = int(s.get("width",  1080))
            h   = int(s.get("height", 1920))
            dur = float(s.get("duration", data["format"].get("duration", 0)))
            break
    return w, h, dur


def _is_vertical(w: int, h: int) -> bool:
    return h > w


def _ffmpeg_base(input_path: str) -> list:
    """Base ffmpeg command with optional GPU decode."""
    cmd = ["ffmpeg", "-y"]
    if _has_gpu:
        cmd += ["-hwaccel", "cuda"]
    cmd += ["-i", input_path]
    return cmd


def _video_codec() -> list:
    """Best available video encoder."""
    if _has_gpu:
        return ["-c:v", "h264_nvenc", "-preset", "p4", "-cq", "22"]
    return ["-c:v", "libx264", "-preset", "fast", "-crf", "22"]


# ─────────────────────────────────────────────────────────────────────────────
#  GEMMA — optional, graceful fallback
# ─────────────────────────────────────────────────────────────────────────────

def _try_gemma(words: list, duration: float, video_path: str):
    """
    Try Gemma analysis. Returns EditDecision or None if Ollama is unavailable/OOM.
    """
    try:
        import requests
        requests.get("http://localhost:11434", timeout=3)  # quick ping
    except Exception:
        log.warning("[gemma] Ollama not reachable — skipping, using smart defaults")
        return None

    try:
        from gemma_brain import analyze
        return analyze(words, duration, video_path=video_path)
    except Exception as e:
        log.warning(f"[gemma] Failed ({e}) — using smart defaults")
        return None


def _smart_defaults(words: list, duration: float):
    """Rule-based fallback when Gemma is unavailable."""
    from dataclasses import dataclass

    # Guess style from word density (words/sec)
    wps = len(words) / max(duration, 1)
    style = "energetic" if wps > 2.5 else "educational"

    # Find most energetic moment (longest word cluster in short window) for hook
    hook = "Yeh dekho!"
    if words:
        transcript = " ".join(w["word"] for w in words[:30])
        hook = transcript[:40].rsplit(" ", 1)[0] + "..."

    # Simple energy map — high in middle
    energy_map = [
        {"time": 0.0,            "energy": "low"},
        {"time": duration * 0.2, "energy": "high"},
        {"time": duration * 0.5, "energy": "peak"},
        {"time": duration * 0.8, "energy": "high"},
    ]

    # Natural cut points at every ~12s
    cut_points = [round(t, 1) for t in
                  list(range(12, int(duration), 12))][:6]

    class Defaults:
        pass

    d = Defaults()
    d.style             = style
    d.hook              = hook
    d.caption_color     = "#FFFF00"
    d.caption_font      = "bold"
    d.cut_points        = cut_points
    d.broll_prompts     = []
    d.energy_map        = energy_map
    d.suggested_trim    = {"start": 0, "end": duration}
    d.scene_description = ""
    d.visual_style      = ""
    return d


# ─────────────────────────────────────────────────────────────────────────────
#  CAPTION BUILDER (Hinglish — pure ASCII/Roman, clean positioning)
# ─────────────────────────────────────────────────────────────────────────────

def _ass_time(t: float) -> str:
    h = int(t // 3600)
    m = int((t % 3600) // 60)
    s = t % 60
    return f"{h}:{m:02d}:{s:05.2f}"


def _build_ass(words: list, highlight_color: str = "#FFFF00",
               font_style: str = "bold",
               video_w: int = 1080, video_h: int = 1920) -> str:
    """
    Word-by-word highlighted captions, Submagic style.
    Positioned at 75% down the screen (above bottom third).
    Font: Arial Bold — handles Hinglish perfectly.
    """
    # Convert hex → ASS colour (&H00BBGGRR)
    hx = highlight_color.lstrip("#")
    if len(hx) == 6:
        r, g, b = int(hx[0:2],16), int(hx[2:4],16), int(hx[4:6],16)
    else:
        r, g, b = 255, 255, 0
    ass_hi = f"&H00{b:02X}{g:02X}{r:02X}"

    font_size = int(22 * video_h / 1920)
    margin_v  = int(video_h * 0.22)   # 22% from bottom = sits nicely above chin area

    header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {video_w}
PlayResY: {video_h}
WrapStyle: 0

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Default,Arial,{font_size},&H00FFFFFF,&H000000FF,&H00000000,&HAA000000,-1,0,0,0,100,100,1,0,1,3,0,2,60,60,{margin_v},1
Style: Hi,Arial,{font_size},{ass_hi},&H000000FF,&H00000000,&HAA000000,-1,0,0,0,100,100,1,0,1,3,0,2,60,60,{margin_v},1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""
    events = []
    WORDS_PER_LINE = 5

    i = 0
    while i < len(words):
        group    = words[i: i + WORDS_PER_LINE]
        group_end = group[-1]["end"]

        for j, w in enumerate(group):
            seg_start = w["start"]
            seg_end   = group[j+1]["start"] if j+1 < len(group) else group_end

            # Build the line — highlighted word + dimmed rest
            parts = []
            for k, gw in enumerate(group):
                txt = gw["word"].strip().replace("{","").replace("}","")
                if not txt:
                    continue
                if k == j:
                    parts.append(f"{{\\rHi}}{txt}{{\\rDefault}}")
                else:
                    parts.append(txt)

            line = " ".join(parts)
            events.append(
                f"Dialogue: 0,{_ass_time(seg_start)},{_ass_time(seg_end)},"
                f"Default,,0,0,0,,{line}"
            )
        i += WORDS_PER_LINE

    return header + "\n".join(events) + "\n"


# ─────────────────────────────────────────────────────────────────────────────
#  MAIN PIPELINE
# ─────────────────────────────────────────────────────────────────────────────

def run_pipeline(input_path: str, output_path: str) -> dict:
    work = Path(tempfile.mkdtemp(prefix="shorts_"))
    log.info(f"[pipeline] Work: {work}")

    src_w, src_h, duration = _probe(input_path)
    already_vertical = _is_vertical(src_w, src_h)
    log.info(f"[pipeline] Source: {src_w}x{src_h} ({duration:.1f}s) "
             f"| vertical={already_vertical}")

    # ── 1. Whisper — Hinglish transcription ─────────────────────────────────
    # Strategy: detect language first, then force transcribe (not translate)
    # For Hinglish, we detect as "hi" (Hindi) but transcribe with Roman prompt
    # This prevents translation while keeping Roman script output
    log.info("[pipeline] Step 1: Detecting language...")
    audio = whisper_lib.load_audio(input_path)
    audio_clip = whisper_lib.pad_or_trim(audio)
    mel = whisper_lib.log_mel_spectrogram(audio_clip).to(_device)
    _, probs = _whisper.detect_language(mel)
    detected_lang = max(probs, key=probs.get)
    log.info(f"[pipeline] Detected language: {detected_lang} ({probs[detected_lang]:.2f})")

    # If Hindi detected, transcribe as Hindi but use initial_prompt to force Roman
    # If English detected, transcribe normally
    # Either way: task=transcribe, NEVER translate
    log.info("[pipeline] Step 1: Whisper transcription (Hinglish mode)")
    # word_timestamps=True needs Triton/CUDA toolkit for DTW alignment
    # Since CUDA toolkit isn't installed, use segment timestamps instead
    # Captions will be per-phrase (5-8 words) rather than per-word — still looks great
    result = _whisper.transcribe(
        input_path,
        task="transcribe",
        language=detected_lang,
        initial_prompt=(
            "Yaar dekho, aisa hota h. Architecture mein kya difference h, "
            "samjhte ho? Bilkul sahi kaha. Toh basically kya h ki, "
            "ek dum clear h yeh concept. Matlab simply bolo toh."
        ),
        word_timestamps=False,    # avoids Triton/DTW — uses segment timestamps
        verbose=False,
        condition_on_previous_text=True,
    )
    # Build word list from segments (word_timestamps=False gives segment-level timing)
    # We split each segment text into individual words and distribute time evenly
    # This gives smooth per-word caption highlighting without needing Triton
    words = []
    for seg in result["segments"]:
        seg_text  = seg["text"].strip()
        seg_start = seg["start"]
        seg_end   = seg["end"]
        seg_words = [w for w in seg_text.split() if w]
        if not seg_words:
            continue
        dur_per_word = (seg_end - seg_start) / len(seg_words)
        for i, w in enumerate(seg_words):
            words.append({
                "word":  w,
                "start": seg_start + i * dur_per_word,
                "end":   seg_start + (i + 1) * dur_per_word,
            })
    transcript_preview = " ".join(w["word"] for w in words[:20])
    log.info(f"[pipeline] Transcribed {len(words)} words")
    log.info(f"[pipeline] Transcript preview: {transcript_preview}")
    # If you see English here but spoke Hinglish, Whisper is translating
    # Check: task should be 'transcribe' not 'translate'
    print(f"\n=== TRANSCRIPT PREVIEW ===\n{transcript_preview}\n==========================\n")

    # ── 2. Gemma (optional) ──────────────────────────────────────────────────
    log.info("[pipeline] Step 2: Gemma analysis (optional)")
    decision = _try_gemma(words, duration, input_path)
    if decision is None:
        log.info("[pipeline] Using rule-based defaults")
        decision = _smart_defaults(words, duration)

    # ── 3. Build captions ────────────────────────────────────────────────────
    log.info("[pipeline] Step 3: Building ASS captions")
    # For vertical source use its actual dims; for landscape use 1080x1920 target
    cap_w = src_w if already_vertical else 1080
    cap_h = src_h if already_vertical else 1920

    ass_content = _build_ass(words, decision.caption_color,
                              decision.caption_font, cap_w, cap_h)
    ass_path = str(work / "captions.ass")
    with open(ass_path, "w", encoding="utf-8") as f:
        f.write(ass_content)

    # ── 4. Trim ───────────────────────────────────────────────────────────────
    trim_start = float(decision.suggested_trim.get("start", 0))
    trim_end   = float(decision.suggested_trim.get("end", duration))
    trimmed    = str(work / "trimmed.mp4")
    _run(["ffmpeg", "-y",
          "-ss", trim_start, "-to", trim_end,
          "-i", input_path,
          "-c", "copy", trimmed], "trim")

    # ── 5. Crop/resize — ONLY if not already vertical ─────────────────────────
    if already_vertical:
        log.info("[pipeline] Step 5: Source is already 9:16 — skipping crop")
        cropped = trimmed
        out_w, out_h = src_w, src_h
    else:
        log.info("[pipeline] Step 5: Cropping landscape → 9:16")
        cropped   = str(work / "cropped.mp4")
        crop_w_px = int(src_h * 9 / 16)
        crop_vf   = (f"crop={crop_w_px}:{src_h}:(iw-{crop_w_px})/2:0,"
                     f"scale=1080:1920:flags=lanczos")
        _run(_ffmpeg_base(trimmed) + [
            "-vf", crop_vf,
            *_video_codec(),
            "-c:a", "copy", cropped
        ], "crop")
        out_w, out_h = 1080, 1920

    # ── 6. Effects + captions ─────────────────────────────────────────────────
    log.info("[pipeline] Step 6: Effects + captions")
    effected = str(work / "effected.mp4")

    # Colour grade based on style
    from effects import color_grade_filter, zoom_punch_filter, hook_drawtext_filter
    grade = color_grade_filter(decision.style)

    # Zoom punch — gentle, based on energy map
    fps   = 30
    zoom  = zoom_punch_filter(decision.energy_map, fps)

    # Captions — escape path for FFmpeg
    ass_esc  = ass_path.replace("\\", "/").replace(":", "\\:")
    captions = f"ass='{ass_esc}'"

    # Hook text overlay
    hook_f = hook_drawtext_filter(decision.hook)

    vf_parts = [grade, zoom, captions]
    if hook_f and hook_f != "null":
        vf_parts.append(hook_f)
    vf_chain = ",".join(vf_parts)

    af_chain = "loudnorm=I=-14:TP=-2:LRA=7,equalizer=f=3000:width_type=o:width=2:g=3"

    _run(_ffmpeg_base(cropped) + [
        "-vf", vf_chain,
        "-af", af_chain,
        *_video_codec(),
        "-c:a", "aac", "-b:a", "192k",
        "-movflags", "+faststart",
        effected
    ], "effects")

    # ── 7. B-roll (only if Gemma gave prompts) ────────────────────────────────
    current = effected
    broll_items = []
    if decision.broll_prompts:
        try:
            from image_gen import generate_all_broll
            broll_items = generate_all_broll(decision.broll_prompts, str(work))
            log.info(f"[pipeline] Generated {len(broll_items)} b-roll frames")
            # (concat logic unchanged from before — skipped here for brevity
            #  since Gemma is optional; b-roll only fires when Gemma is up)
        except Exception as e:
            log.warning(f"[pipeline] B-roll skipped: {e}")

    # ── 8. Final export ───────────────────────────────────────────────────────
    log.info("[pipeline] Step 8: Final export")
    _run(["ffmpeg", "-y", "-i", current,
          "-c", "copy", "-movflags", "+faststart",
          output_path], "final")

    log.info(f"[pipeline] Done → {output_path}")

    try:
        shutil.rmtree(work)
    except Exception:
        pass

    return {
        "style":             decision.style,
        "hook":              decision.hook,
        "n_cuts":            len(decision.cut_points),
        "n_broll":           len(broll_items),
        "scene_description": getattr(decision, "scene_description", ""),
        "visual_style":      getattr(decision, "visual_style", ""),
        "used_gemma":        decision.__class__.__name__ != "Defaults",
    }
